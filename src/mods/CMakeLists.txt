# Mods: every folder here holding a CMakeLists.txt is one (cmake/GdlMods.cmake). Each one found
# is compiled into the programs, and the generated list below registers its mechanics at
# startup; the catalog orders them by id, whatever the order here.
include(GdlMods)

file(GLOB GDL_MOD_FOLDERS LIST_DIRECTORIES true CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*")
list(SORT GDL_MOD_FOLDERS)
foreach(folder IN LISTS GDL_MOD_FOLDERS)
    if(IS_DIRECTORY "${folder}" AND EXISTS "${folder}/CMakeLists.txt")
        add_subdirectory("${folder}")
    endif()
endforeach()

get_property(GDL_MOD_NAMES GLOBAL PROPERTY GDL_MODS)
set(GDL_MOD_DECLARATIONS "")
set(GDL_MOD_CALLS "")
foreach(name IN LISTS GDL_MOD_NAMES)
    string(APPEND GDL_MOD_DECLARATIONS
        "namespace gdl::mods::${name} {\n"
        "void registerMechanics(game::MechanicCatalog& catalog);\n"
        "} // namespace gdl::mods::${name}\n\n")
    string(APPEND GDL_MOD_CALLS "    mods::${name}::registerMechanics(catalog);\n")
endforeach()
set(GDL_COMPILED_MECHANICS "${CMAKE_BINARY_DIR}/generated/mods/CompiledMechanics.cpp")
file(CONFIGURE OUTPUT "${GDL_COMPILED_MECHANICS}" @ONLY CONTENT [=[
// Generated by src/mods/CMakeLists.txt from the mods found there; do not edit.
#include "game/mechanics/MechanicCatalog.h"

@GDL_MOD_DECLARATIONS@namespace gdl::game {

void registerCompiledMechanics([[maybe_unused]] MechanicCatalog& catalog) {
@GDL_MOD_CALLS@}

} // namespace gdl::game
]=])

add_library(mods STATIC "${GDL_COMPILED_MECHANICS}")
add_library(gdl::mods ALIAS mods)
target_link_libraries(mods PUBLIC gdl::game)
foreach(name IN LISTS GDL_MOD_NAMES)
    target_link_libraries(mods PUBLIC mod_${name})
endforeach()
gdl_apply_compiler_options(mods)
set_target_properties(mods PROPERTIES FOLDER "mods")
